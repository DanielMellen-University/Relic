# The C runtime for compiled Relic (milestone C1). Plain C99.
# Built as part of the main project, or on its own:
#   cmake -S runtime/c -B build-c && cmake --build build-c && ctest --test-dir build-c
cmake_minimum_required(VERSION 3.16)
if(CMAKE_CURRENT_SOURCE_DIR STREQUAL CMAKE_SOURCE_DIR)
  project(relic_c_runtime LANGUAGES C)
  enable_testing()
else()
  enable_language(C)
endif()
get_filename_component(RELIC_ROOT ${CMAKE_CURRENT_SOURCE_DIR}/../.. ABSOLUTE)

option(RELIC_C_WERROR "Treat C runtime warnings as errors" OFF)
option(RELIC_C_SANITIZE "Build the C runtime with ASan and UBSan (gcc/clang)" OFF)

option(RELIC_C_SDL "Build the SDL2 graphics module (rl_gfx_sdl.c) into the C runtime in place of the stub; needs SDL2" OFF)

if(RELIC_C_SDL)
  set(RL_GFX rl_gfx_sdl.c)
else()
  set(RL_GFX rl_gfx_stub.c)
endif()
set(RL_SOURCES rl_core.c rl_ops.c rl_builtins.c rl_host.c ${RL_GFX})
add_library(relic_rt STATIC ${RL_SOURCES})
target_include_directories(relic_rt PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_compile_definitions(relic_rt PRIVATE RL_STD_DIR="${RELIC_ROOT}/std/")

# Graphics for compiled programs: the same host_gfx_* table as the C++ host
# (src/builtins/graphics.cpp). SDL2_image and SDL2_ttf are used when found.
if(RELIC_C_SDL)
  find_package(SDL2 REQUIRED)
  if(TARGET SDL2::SDL2)
    target_link_libraries(relic_rt PUBLIC SDL2::SDL2)
  else()
    target_include_directories(relic_rt PRIVATE ${SDL2_INCLUDE_DIRS})
    target_link_libraries(relic_rt PUBLIC ${SDL2_LIBRARIES})
  endif()
  # The built-in 8x12 font is shared with the host.
  target_include_directories(relic_rt PRIVATE ${RELIC_ROOT}/src/builtins)
  find_package(SDL2_image CONFIG QUIET)
  if(TARGET SDL2_image::SDL2_image)
    target_compile_definitions(relic_rt PRIVATE RL_HAVE_SDL_IMAGE)
    target_link_libraries(relic_rt PUBLIC SDL2_image::SDL2_image)
    set(RL_C_SDL_IMAGE ON)
    message(STATUS "Relic C graphics: PNG images on (SDL2_image)")
  endif()
  find_package(SDL2_ttf CONFIG QUIET)
  if(TARGET SDL2_ttf::SDL2_ttf)
    target_compile_definitions(relic_rt PRIVATE RL_HAVE_SDL_TTF)
    target_link_libraries(relic_rt PUBLIC SDL2_ttf::SDL2_ttf)
    set(RL_C_SDL_TTF ON)
    message(STATUS "Relic C graphics: TrueType fonts on (SDL2_ttf)")
  endif()
  message(STATUS "Relic C graphics: SDL2 module on")
endif()
# What the module was built with, for the tests in the top-level CMakeLists.
set(RELIC_C_SDL_IMAGE ${RL_C_SDL_IMAGE} CACHE INTERNAL "C graphics have SDL2_image")
set(RELIC_C_SDL_TTF ${RL_C_SDL_TTF} CACHE INTERNAL "C graphics have SDL2_ttf")

# Strict flags for runtime code and for C generated by cgen. Also called from
# the top-level CMakeLists for the cgen_* tests.
function(relic_c_target t)
  # MSVC has no C99 switch; C11 mode is the closest and takes C99 code.
  if(MSVC)
    set(std 11)
  else()
    set(std 99)
  endif()
  set_target_properties(${t} PROPERTIES C_STANDARD ${std} C_STANDARD_REQUIRED ON C_EXTENSIONS OFF)
  if(MSVC)
    target_compile_options(${t} PRIVATE /W4)
    target_compile_definitions(${t} PRIVATE _CRT_SECURE_NO_WARNINGS)
    if(RELIC_C_WERROR)
      target_compile_options(${t} PRIVATE /WX)
    endif()
  else()
    target_compile_options(${t} PRIVATE -Wall -Wextra -pedantic)
    if(RELIC_C_WERROR)
      target_compile_options(${t} PRIVATE -Werror)
    endif()
    if(RELIC_C_SANITIZE)
      target_compile_options(${t} PRIVATE -fsanitize=address,undefined -fno-sanitize-recover=undefined -fno-omit-frame-pointer)
      target_link_options(${t} PRIVATE -fsanitize=address,undefined)
    endif()
  endif()
endfunction()

relic_c_target(relic_rt)
if(NOT WIN32)
  target_link_libraries(relic_rt PUBLIC m)
endif()
